-- 1. Kiểm tra và tạo Database nếu chưa tồn tại
IF NOT EXISTS (SELECT * FROM sys.databases WHERE name = 'KiemKeDb')
BEGIN
    CREATE DATABASE KiemKeDb;
END
GO

USE KiemKeDb;
GO

-- 2. Bảng Phân Xưởng
IF NOT EXISTS (SELECT 1 FROM sys.tables WHERE name = 'Workshops')
BEGIN
    CREATE TABLE Workshops (
        Id INT IDENTITY(1,1) PRIMARY KEY,
        Name NVARCHAR(100) NOT NULL UNIQUE,
        Password NVARCHAR(50) NOT NULL
    );
END
GO

-- 3. Bảng Danh Mục Vật Tư (Độc lập theo Phân xưởng)
IF NOT EXISTS (SELECT 1 FROM sys.tables WHERE name = 'Categories')
BEGIN
    CREATE TABLE Categories (
        Id INT IDENTITY(1,1) PRIMARY KEY,
        Name NVARCHAR(100) NOT NULL,
        Workshop NVARCHAR(100) NOT NULL
    );
END
GO

-- 4. Bảng Máy / Vị Trí (Độc lập theo Phân xưởng)
IF NOT EXISTS (SELECT 1 FROM sys.tables WHERE name = 'Machines')
BEGIN
    CREATE TABLE Machines (
        Id INT IDENTITY(1,1) PRIMARY KEY,
        Name NVARCHAR(100) NOT NULL,
        Workshop NVARCHAR(100) NOT NULL
    );
END
GO

-- 5. Bảng Master Data Mã SAP (Hỗ trợ Aliases và phân tách riêng cho từng Phân xưởng)
IF NOT EXISTS (SELECT 1 FROM sys.tables WHERE name = 'Materials')
BEGIN
    CREATE TABLE Materials (
        Id INT IDENTITY(1,1) PRIMARY KEY,
        SapCode NVARCHAR(50) NOT NULL,
        Name NVARCHAR(250) NOT NULL,
        Description NVARCHAR(250),
        Category NVARCHAR(100) NOT NULL,
        Workshop NVARCHAR(100) NOT NULL,
        Aliases NVARCHAR(500)
    );
END
GO

-- Đảm bảo cột Aliases luôn tồn tại nếu database đã có sẵn bảng này từ trước
IF NOT EXISTS (
    SELECT 1 FROM sys.columns 
    WHERE object_id = OBJECT_ID(N'[dbo].[Materials]') AND name = 'Aliases'
)
BEGIN
    ALTER TABLE [dbo].[Materials] ADD [Aliases] NVARCHAR(500) NULL;
END
GO

-- 6. Bảng Phiếu Kiểm Kê
IF NOT EXISTS (SELECT 1 FROM sys.tables WHERE name = 'InventorySheets')
BEGIN
    CREATE TABLE InventorySheets (
        Id NVARCHAR(50) PRIMARY KEY,
        Workshop NVARCHAR(100) NOT NULL,
        DeviceId NVARCHAR(100),
        Date NVARCHAR(20) NOT NULL,
        Machine NVARCHAR(100),
        CreatedBy NVARCHAR(100) NOT NULL,
        CreatedAt DATETIME DEFAULT GETDATE()
    );
END
GO

-- 7. Bảng Chi Tiết Vật Tư Trong Phiếu Kiểm Kê
IF NOT EXISTS (SELECT 1 FROM sys.tables WHERE name = 'InventorySheetItems')
BEGIN
    CREATE TABLE InventorySheetItems (
        Id INT IDENTITY(1,1) PRIMARY KEY,
        SheetId NVARCHAR(50) NOT NULL FOREIGN KEY REFERENCES InventorySheets(Id) ON DELETE CASCADE,
        SapCode NVARCHAR(50) NOT NULL,
        Name NVARCHAR(250) NOT NULL,
        Category NVARCHAR(100),
        Workshop NVARCHAR(100),
        Machine NVARCHAR(100),
        Quantity DECIMAL(18,2) DEFAULT 0,
        Tubes INT DEFAULT 0,
        Note NVARCHAR(250)
    );
END
GO
